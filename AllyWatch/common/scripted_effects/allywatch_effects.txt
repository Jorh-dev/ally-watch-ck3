# Every effect bails out when nobody is watching the character, so alliance churn
# across the rest of the map costs one list check.
#
# Notification thresholds live on the WATCHING PLAYER, not on the watched ruler,
# so two players can watch the same person with different settings. Script has no
# map type, so one variable list per threshold value stands in for the key-value
# store: being in allywatch_thr_3 means "tell me once they are down to 3".
# All threshold effects below run in the watching player's scope.

# Removes any threshold this player set for the target. Shared cleanup step for
# changing a threshold, dropping a ruler, and death.
allywatch_clear_threshold_effect = {
	# $TARGET$ = watched character
	remove_list_variable = { name = allywatch_thr_0 target = $TARGET$ }
	remove_list_variable = { name = allywatch_thr_1 target = $TARGET$ }
	remove_list_variable = { name = allywatch_thr_2 target = $TARGET$ }
	remove_list_variable = { name = allywatch_thr_3 target = $TARGET$ }
	remove_list_variable = { name = allywatch_thr_4 target = $TARGET$ }
	remove_list_variable = { name = allywatch_thr_5 target = $TARGET$ }
	remove_list_variable = { name = allywatch_thr_6 target = $TARGET$ }
	remove_list_variable = { name = allywatch_thr_7 target = $TARGET$ }
	remove_list_variable = { name = allywatch_thr_8 target = $TARGET$ }
}

# Clear first, then file under the new level, so the target is never in two lists.
allywatch_set_threshold_effect = {
	# $TARGET$ = watched character, $LEVEL$ = 0..8
	allywatch_clear_threshold_effect = { TARGET = $TARGET$ }
	add_to_variable_list = {
		name = allywatch_thr_$LEVEL$
		target = $TARGET$
	}
}

# The ladder is 0 (only when alone) up to 8, plus "no threshold" as a separate state
# reached in one click rather than by walking the whole ladder: setting a threshold
# lands straight on 0, which is what people actually want, and clearing it is also
# one click. Stepping is only for the rarely-needed values in between.
# List names cannot be built from a computed number, so each rung is spelled out.
# else_if means exactly one branch runs.

# One rung up: report earlier, while they still have allies left.
allywatch_threshold_up_effect = {
	# $TARGET$ = watched character
	if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_7 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 8 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_6 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 7 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_5 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 6 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_4 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 5 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_3 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 4 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_2 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 3 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_1 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 2 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_0 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 1 }
	}
}

# One rung down: hold off until they are further down.
# thr_0 is the bottom, so there it does nothing and the option is hidden.
allywatch_threshold_down_effect = {
	# $TARGET$ = watched character
	if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_8 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 7 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_7 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 6 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_6 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 5 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_5 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 4 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_4 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 3 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_3 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 2 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_2 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 1 }
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = allywatch_thr_1 target = $TARGET$ } }
		allywatch_set_threshold_effect = { TARGET = $TARGET$ LEVEL = 0 }
	}
}

# Drops a ruler from this player's watch list and forgets their threshold.
allywatch_untrack_effect = {
	# $TARGET$ = watched character
	remove_list_variable = {
		name = allywatch_tracked
		target = $TARGET$
	}
	allywatch_clear_threshold_effect = { TARGET = $TARGET$ }
}

# Stamps the watched character so the same event is not reported twice.
allywatch_mark_change_effect = {
	# $WATCHED$ / $OTHER$ / $VARNAME$
	$WATCHED$ = {
		set_variable = {
			name = $VARNAME$
			value = $OTHER$
			days = 1
		}
	}
}

# The watched character lost an ally. If it was their last one, send the louder message.
allywatch_notify_ally_lost_effect = {
	# $WATCHED$ = tracked candidate, $OTHER$ = the ally they just lost
	if = {
		limit = {
			allywatch_is_tracked_trigger = { CHARACTER = $WATCHED$ }
			allywatch_is_new_change_trigger = {
				WATCHED = $WATCHED$
				OTHER = $OTHER$
				VARNAME = allywatch_last_lost
			}
		}
		allywatch_mark_change_effect = {
			WATCHED = $WATCHED$
			OTHER = $OTHER$
			VARNAME = allywatch_last_lost
		}
		# The ally count in the message text comes from a customizable_localization on
		# LEFT_CHARACTER, not from a saved scope: a message is rendered after the effect
		# has finished, so nothing saved here would still be around.
		every_player = {
			limit = {
				is_target_in_variable_list = {
					name = allywatch_tracked
					target = $WATCHED$
				}
			}
			# Losing the last ally is always reported. The intermediate "lost one"
			# message is the noisy one, so that is what the threshold silences.
			if = {
				limit = {
					$WATCHED$ = { NOT = { any_ally = { always = yes } } }
				}
				send_interface_message = {
					type = allywatch_msg_no_allies
					left_icon = $WATCHED$
					right_icon = $OTHER$
				}
			}
			# Thresholds are per player, so this is checked inside the player loop.
			else_if = {
				limit = {
					allywatch_threshold_allows_trigger = { WATCHED = $WATCHED$ }
				}
				send_interface_message = {
					type = allywatch_msg_ally_lost
					left_icon = $WATCHED$
					right_icon = $OTHER$
				}
			}
		}
		allywatch_record_ex_ally_effect = {
			WATCHED = $WATCHED$
			OTHER = $OTHER$
		}
	}
}

# Feeds the "broken alliances" section of the report. Only called for tracked
# characters, so nothing is written to rulers nobody watches.
allywatch_record_ex_ally_effect = {
	# $WATCHED$ = tracked character, $OTHER$ = the ally they just lost
	$WATCHED$ = {
		# Remove first so a repeat break moves the entry instead of duplicating it.
		remove_list_variable = {
			name = allywatch_ex_allies
			target = $OTHER$
		}
		add_to_variable_list = {
			name = allywatch_ex_allies
			target = $OTHER$
		}
		# Cap the history at 5 so the list cannot grow without bound over a long game.
		while = {
			limit = {
				variable_list_size = {
					name = allywatch_ex_allies
					value > 5
				}
			}
			ordered_in_list = {
				variable = allywatch_ex_allies
				position = 0
				save_temporary_scope_as = allywatch_oldest_ex_ally
			}
			remove_list_variable = {
				name = allywatch_ex_allies
				target = scope:allywatch_oldest_ex_ally
			}
		}
	}
}

# The watched character gained an ally - bad news for whoever is watching them.
allywatch_notify_ally_gained_effect = {
	# $WATCHED$ = tracked candidate, $OTHER$ = their new ally
	if = {
		limit = {
			allywatch_is_tracked_trigger = { CHARACTER = $WATCHED$ }
			allywatch_is_new_change_trigger = {
				WATCHED = $WATCHED$
				OTHER = $OTHER$
				VARNAME = allywatch_last_gained
			}
		}
		allywatch_mark_change_effect = {
			WATCHED = $WATCHED$
			OTHER = $OTHER$
			VARNAME = allywatch_last_gained
		}
		every_player = {
			limit = {
				is_target_in_variable_list = {
					name = allywatch_tracked
					target = $WATCHED$
				}
			}
			send_interface_message = {
				type = allywatch_msg_ally_gained
				left_icon = $WATCHED$
				right_icon = $OTHER$
			}
		}
		# Allied again: drop them from the broken-alliance history so the report
		# never lists the same person as both a current and a former ally.
		$WATCHED$ = {
			remove_list_variable = {
				name = allywatch_ex_allies
				target = $OTHER$
			}
		}
	}
}

# Status panel. The message type uses desc = event_message_effect, so what this
# block renders becomes the popup body. show_as_tooltip renders without executing,
# so the loops print a line per ally and change nothing.
allywatch_show_report_effect = {
	# $TARGET$ = the ruler being inspected
	$TARGET$ = { save_scope_as = allywatch_report_target }
	# The reader is skipped in both lists below: being one of the target's own allies,
	# they would otherwise turn up as a line reading "you", and the game groups lines
	# by whose scope they are in, so that one line dragged the reader's portrait to the
	# top of a panel that is supposed to be about somebody else.
	scope:actor = { save_scope_as = allywatch_reader }
	scope:actor = {
		send_interface_message = {
			type = allywatch_msg_report
			left_icon = scope:allywatch_report_target
			# Everything below runs inside the WATCHED ruler's scope, never the reader's.
			# show_as_tooltip renders each line under a portrait of whichever character
			# its scope is in, so a header written out here came out under the reader's
			# own face with vanilla's "effects on me" caption - the panel appeared to be
			# about the player rather than the ruler they had just right-clicked.
			show_as_tooltip = {
				scope:allywatch_report_target = {
					if = {
						limit = { any_ally = { always = yes } }
						custom_tooltip = allywatch_report_allies_header
						# Inside the loop the line's loc reads THIS, which is the
						# current ally - same pattern vanilla uses for MIGRATION_CB_TITLE.
						every_ally = {
							limit = { NOT = { this = scope:allywatch_reader } }
							custom_tooltip = allywatch_report_ally_line
						}
						# The reader is left out of the loop so their own portrait cannot
						# hijack the panel, but the fact itself matters - being on the list
						# of people this ruler can call in changes what a war costs you.
						if = {
							limit = { is_allied_to = scope:allywatch_reader }
							custom_tooltip = allywatch_report_ally_is_you
						}
					}
					else = {
						custom_tooltip = allywatch_report_no_allies
					}

					# Only populated for characters that were on a watch list when the
					# alliance broke, so this section is simply absent for everyone else.
					# has_variable_list first: a list never written to does not exist,
					# and asking for its size is an error rather than a zero.
					if = {
						limit = {
							has_variable_list = allywatch_ex_allies
							variable_list_size = {
								name = allywatch_ex_allies
								value > 0
							}
						}
						custom_tooltip = allywatch_report_broken_header
						every_in_list = {
							variable = allywatch_ex_allies
							limit = { NOT = { this = scope:allywatch_reader } }
							custom_tooltip = allywatch_report_broken_line
						}
						if = {
							limit = {
								is_target_in_variable_list = {
									name = allywatch_ex_allies
									target = scope:allywatch_reader
								}
							}
							custom_tooltip = allywatch_report_broken_is_you
						}
					}
				}
			}
		}
	}
}

# Called from on_death only, where root is the dying character.
# Reports the death, names the successor, then drops the dead character from
# every watch list so no variable list keeps pointing at a corpse.
allywatch_notify_death_effect = {
	if = {
		limit = {
			allywatch_is_tracked_trigger = { CHARACTER = root }
		}
		save_scope_as = allywatch_dead
		if = {
			limit = { exists = primary_heir }
			primary_heir = { save_scope_as = allywatch_heir }
		}
		every_player = {
			limit = {
				is_target_in_variable_list = {
					name = allywatch_tracked
					target = scope:allywatch_dead
				}
			}
			if = {
				limit = { exists = scope:allywatch_heir }
				send_interface_message = {
					type = allywatch_msg_death
					left_icon = scope:allywatch_dead
					right_icon = scope:allywatch_heir
				}
			}
			else = {
				send_interface_message = {
					type = allywatch_msg_death_no_heir
					left_icon = scope:allywatch_dead
				}
			}
			allywatch_untrack_effect = { TARGET = scope:allywatch_dead }
		}
	}
}

# --- Succession -------------------------------------------------------------
# Every list in this file lives on the player's CURRENT character. When a player
# dies the game hands them their heir but leaves the variables behind on the
# corpse, so without this the whole watch list and every threshold quietly vanish
# once a generation - and because allywatch_list_interaction is only shown while
# the list is non-empty, the menu entry disappears with them, leaving no trace
# that anything was lost.
#
# player_heir is what vanilla's own on_death uses for the character a player
# continues as (see common/on_action/death.txt).

allywatch_carry_one_list_effect = {
	# $LIST$ = variable list to carry over, $TO$ = scope of the successor
	# A list never written to does not exist at all, hence the guard.
	if = {
		limit = { has_variable_list = $LIST$ }
		every_in_list = {
			variable = $LIST$
			save_temporary_scope_as = allywatch_carried
			$TO$ = {
				add_to_variable_list = {
					name = $LIST$
					target = scope:allywatch_carried
				}
			}
		}
	}
}

# Called from on_death, where root is the dying character.
# Thresholds travel with the watch list; a carried-over ruler the player had
# silenced should stay silenced.
allywatch_carry_lists_to_heir_effect = {
	if = {
		limit = {
			is_ai = no
			exists = player_heir
			allywatch_is_watching_trigger = yes
		}
		player_heir = { save_scope_as = allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_tracked TO = scope:allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_thr_0 TO = scope:allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_thr_1 TO = scope:allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_thr_2 TO = scope:allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_thr_3 TO = scope:allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_thr_4 TO = scope:allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_thr_5 TO = scope:allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_thr_6 TO = scope:allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_thr_7 TO = scope:allywatch_successor }
		allywatch_carry_one_list_effect = { LIST = allywatch_thr_8 TO = scope:allywatch_successor }
	}
}
