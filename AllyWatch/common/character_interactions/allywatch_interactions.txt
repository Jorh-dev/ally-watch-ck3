# Portrait right-click menu. Free, auto-accepted, recipient is never notified.

allywatch_track_interaction = {
	category = interaction_category_diplomacy
	icon = alliance
	interface_priority = 50

	desc = allywatch_track_interaction_desc

	use_diplomatic_range = no			# a target on the far side of the map can be watched
	auto_accept = yes
	popup_on_receive = no
	needs_recipient_to_open = no
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = {
			is_ai = no
			is_alive = yes
		}
		scope:actor != scope:recipient
		scope:recipient = {
			is_landed = yes
		}
		NOT = {
			scope:actor = {
				is_target_in_variable_list = {
					name = allywatch_tracked
					target = scope:recipient
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_alive = yes
		}
	}

	on_accept = {
		scope:actor = {
			# Outside the toast - must not depend on the toast being enabled.
			add_to_variable_list = {
				name = allywatch_tracked
				target = scope:recipient
			}
			send_interface_toast = {
				title = allywatch_track_started_toast
				left_icon = scope:recipient
				custom_tooltip = allywatch_track_started_toast_tt
			}
		}
	}
}

# Your own portrait. Hidden entirely while the watch list is empty.
allywatch_list_interaction = {
	category = interaction_category_diplomacy
	icon = council
	interface_priority = 51

	desc = allywatch_list_interaction_desc

	use_diplomatic_range = no
	auto_accept = yes
	popup_on_receive = no
	needs_recipient_to_open = no
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			is_ai = no
			is_alive = yes
			# Guarded trigger: a player who has never watched anyone has no list at
			# all, and asking such a list for its size is an error, not a zero.
			allywatch_is_watching_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
	}

	on_accept = {
		scope:actor = {
			set_variable = { name = allywatch_page value = 0 }
			trigger_event = allywatch.0001
		}
	}
}

# Not gated on tracking - works on any ruler.

	interface_priority = 48

	desc = allywatch_report_interaction_desc

	use_diplomatic_range = no
	auto_accept = yes
	popup_on_receive = no
	needs_recipient_to_open = no
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = {
			is_ai = no
			is_alive = yes
		}
		scope:actor != scope:recipient
		scope:recipient = {
			is_landed = yes
		}
	}
